            const pad = (4 - (u8.length % 4)) % 4;
            const buf = new Uint8Array(u8.length + pad);
            buf.set(u8);
            let out = "";
            for (let i = 0; i < buf.length; i += 4) {
                let v = ((buf[i] << 24) | (buf[i + 1] << 16) | (buf[i + 2] << 8) | buf[i + 3]) >>> 0;
                const c4 = v % 85; v = Math.floor(v / 85);
                const c3 = v % 85; v = Math.floor(v / 85);
                const c2 = v % 85; v = Math.floor(v / 85);
                const c1 = v % 85; v = Math.floor(v / 85);
                const c0 = v % 85;
                out += Z85_ENC[c0] + Z85_ENC[c1] + Z85_ENC[c2] + Z85_ENC[c3] + Z85_ENC[c4];
            }
            return out;
        }

        function decodeZ85ToBytes(str, exactByteLen) {
            const groups = Math.floor(str.length / 5);
            const out = new Uint8Array(groups * 4);
            let outIdx = 0;
            for (let i = 0; i < groups * 5; i += 5) {
                const v = (
                    Z85_DEC[str.charCodeAt(i)] * 52200625 +
                    Z85_DEC[str.charCodeAt(i + 1)] * 614125 +
                    Z85_DEC[str.charCodeAt(i + 2)] * 7225 +
                    Z85_DEC[str.charCodeAt(i + 3)] * 85 +
                    Z85_DEC[str.charCodeAt(i + 4)]
                ) >>> 0;
                out[outIdx++] = (v >>> 24) & 0xff;
                out[outIdx++] = (v >>> 16) & 0xff;
                out[outIdx++] = (v >>> 8) & 0xff;
                out[outIdx++] = v & 0xff;
            }
            return exactByteLen !== undefined ? out.subarray(0, exactByteLen) : out;
        }

        // Empaqueta un encabezado binario de 4 bytes [id_hi, id_lo, lenFlags_hi, lenFlags_lo] seguido del payload binario puro
        function packBinaryDroplet(dropletId, sliceBytes, isFountainParity = false) {
            const len = sliceBytes.byteLength;
            const flagsAndLen = (isFountainParity ? 0x8000 : 0x0000) | (len & 0x7FFF);
            const frame = new Uint8Array(4 + len);
            frame[0] = (dropletId >>> 8) & 0xff;
            frame[1] = dropletId & 0xff;
            frame[2] = (flagsAndLen >>> 8) & 0xff;
            frame[3] = flagsAndLen & 0xff;
            frame.set(sliceBytes, 4);
            return "~" + encodeBytesToZ85(frame);
        }

        // PRNG determinista Mulberry32 idéntico al del Receptor para Códigos Fountain (LT XOR)
        function mulberry32(seed) {
            let a = seed >>> 0;
            return function () {
                let t = a += 0x6D2B79F5;
                t = Math.imul(t ^ (t >>> 15), t | 1);
                t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
                return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
            };
        }

        function getFountainRecipe(dropletId, totalChunks) {
            const rng = mulberry32(dropletId);
            const r = rng();
            let degree = 2;
            if (r < 0.50) degree = 2;
            else if (r < 0.75) degree = 3;
            else if (r < 0.90) degree = 4;
            else degree = 6;

            degree = Math.min(degree, totalChunks);
            const chosen = new Set();
            while (chosen.size < degree) {
                chosen.add(Math.floor(rng() * totalChunks));
            }
            return Array.from(chosen);
        }

        function generateFountainParityFrame(seed) {
            const recipe = getFountainRecipe(seed, sys.totalChunks);
            const xorBuf = new Uint8Array(sys.mtu);
            for (const idx of recipe) {
                const src = sys.rawSlices[idx];
                if (src) {
                    for (let b = 0; b < src.length; b++) {
                        xorBuf[b] ^= src[b];
                    }
                }
            }
            return {
                payload: packBinaryDroplet(seed, xorBuf, true),
                recipe
            };
        }

        async function compressGzip(uint8Data) {
            if (typeof CompressionStream === 'undefined') return { data: uint8Data, compressed: 0 };
            try {
                const cs = new CompressionStream('gzip');
                const writer = cs.writable.getWriter();
                writer.write(uint8Data);
                writer.close();
                const compressedBuf = await new Response(cs.readable).arrayBuffer();
                const compressedU8 = new Uint8Array(compressedBuf);
                if (compressedU8.byteLength < uint8Data.byteLength) {
                    return { data: compressedU8, compressed: 1 };
                }
                return { data: uint8Data, compressed: 0 };
            } catch (e) {
                return { data: uint8Data, compressed: 0 };
            }
        }

        function formatBytes(b) {
            if (b < 1024) return b + " B";
            if (b < 1048576) return (b / 1024).toFixed(2) + " KB";
            return (b / 1048576).toFixed(2) + " MB";
        }

        function indicesToRangesString(indicesArr) {
            if (!indicesArr || indicesArr.length === 0) return "";
            const sorted = Array.from(new Set(indicesArr)).sort((a, b) => a - b);
            const ranges = [];
            let start = sorted[0], prev = sorted[0];
            for (let i = 1; i <= sorted.length; i++) {
                if (sorted[i] === prev + 1) {
                    prev = sorted[i];
                } else {
                    ranges.push(start === prev ? `${start}` : `${start}-${prev}`);
                    start = sorted[i];
                    prev = sorted[i];
                }
            }
            return ranges.join(', ');
        }

        async function handleBuffer(blob, filename) {
            if (sys.running) togglePlay();

            if (blob.size === 0) {
                alert("El archivo o texto está vacío.");
                return;
            }

            sys.sourceBlob = blob;
            sys.sourceName = filename;

            const loader = document.getElementById('async-loader');
            const loaderLbl = document.getElementById('async-loader-lbl');
            loader.classList.add('active');
            loaderLbl.innerText = "Calculando firma criptográfica SHA-256...";
            document.getElementById('status').innerText = "● PROCESANDO...";

            await new Promise(r => setTimeout(r, 25));

            try {
                const origBuf = await blob.arrayBuffer();
                const origU8 = new Uint8Array(origBuf);

                const hashArr = await crypto.subtle.digest('SHA-256', origBuf);
                const hash = Array.from(new Uint8Array(hashArr))
                    .map(b => b.toString(16).padStart(2, '0'))
                    .join('').substring(0, 12);

                const useComp = document.getElementById('chk-gzip').checked;
                let finalU8 = origU8;
                let compFlag = 0;

                if (useComp) {
                    loaderLbl.innerText = "Comprimiendo flujo estructurado...";
                    await new Promise(r => setTimeout(r, 20));
                    const res = await compressGzip(origU8);
                    finalU8 = res.data;
                    compFlag = res.compressed;
                }

                loaderLbl.innerText = "Empaquetando tramas binarias 8-bit con encabezado de 4 bytes...";
                await new Promise(r => setTimeout(r, 15));

                const ratio = ((1 - finalU8.byteLength / origU8.byteLength) * 100).toFixed(1);
                document.getElementById('comp-badge').innerText = compFlag
                    ? `8-BIT + COMP (-${ratio}%)`
                    : `8-BIT DIRECTO`;

                const total = Math.ceil(finalU8.byteLength / sys.mtu);
                sys.totalChunks = total;
                sys.fountainSeed = Math.max(total + 100, 10000);

                sys.headerPayload = JSON.stringify({
                    t: "H",
                    c: total,
                    m: sys.mtu,
                    cl: finalU8.byteLength,
                    n: filename,
                    h: hash,
                    s: origU8.byteLength,
                    z: compFlag
                });

                sys.rawSlices = new Array(total);
                sys.chunksPayloads = new Array(total);
                for (let i = 0; i < total; i++) {
                    const slice = finalU8.subarray(i * sys.mtu, (i + 1) * sys.mtu);
                    sys.rawSlices[i] = slice;
                    sys.chunksPayloads[i] = packBinaryDroplet(i, slice, false);
                }

                document.getElementById('mt-name').innerText = filename;
                document.getElementById('mt-size').innerText = formatBytes(origU8.byteLength);
                document.getElementById('mt-comp').innerText = formatBytes(finalU8.byteLength);
                document.getElementById('mt-chunks').innerText = total;
                document.getElementById('mt-hash').innerText = hash;

                sys.qr = new QRCode(document.getElementById("qr-canvas-mount"), {
                    width: 640,
                    height: 640,
                    correctLevel: QRCode.CorrectLevel.L
                });

                sys.activeIndices = Array.from({ length: total }, (_, i) => i);
                sys.activeSet = new Set(sys.activeIndices);
                sys.activeIdx = 0;
                sys.tickCount = 0;
                setTxMode('full', true);

                ['run-btn', 'btn-rewind', 'btn-prev', 'btn-next', 'btn-zen', 'btn-scan-ack', 'timeline-scrubber'].forEach(id => {
                    document.getElementById(id).disabled = false;
                });

                updateTimelineBounds();
                renderHeaderFrame();
                document.getElementById('status').innerText = "● PAYLOAD 8-BIT LISTO";
            } catch (err) {
                console.error("Error procesando buffer:", err);
                alert("Error al empaquetar datos: " + err.message);
            } finally {
                loader.classList.remove('active');
            }
        }

        function draw2DGridSelector() {
            const canvas = document.getElementById('grid-2d-selector');
            if (!canvas) return;
            const dpr = window.devicePixelRatio || 1;
            const w = (canvas.clientWidth || 440) * dpr;
            const h = (canvas.clientHeight || 155) * dpr;
            if (canvas.width !== w || canvas.height !== h) {
                canvas.width = w;
                canvas.height = h;
            }
            const ctx = canvas.getContext('2d');
            ctx.clearRect(0, 0, w, h);

            const { total, cols, rows } = getGridGeometry();
            const cellW = w / cols;
            const cellH = h / rows;
            const pad = total > 2500 ? 0 : 0.5;
            const currentChunk = sys.activeIndices.length > 0 ? sys.activeIndices[sys.activeIdx] : -1;

            for (let i = 0; i < total; i++) {
                const c = i % cols;
                const r = Math.floor(i / cols);
                const x = c * cellW;
                const y = r * cellH;
                const isActive = sys.totalChunks > 0 ? sys.activeSet.has(i) : false;

                if (i === currentChunk) {
                    ctx.fillStyle = '#ffffff';
                } else if (sys.txMode === 'priority') {
                    ctx.fillStyle = isActive ? '#ff9e00' : '#0b381a';
                } else if (sys.txMode === 'fountain') {
                    ctx.fillStyle = '#2f81f7';
                } else {
                    ctx.fillStyle = isActive ? '#00b82d' : '#161b22';
                }

                ctx.fillRect(
                    x + pad,
                    y + pad,
                    Math.max(1, cellW - pad * 2),
                    Math.max(1, cellH - pad * 2)
                );
            }

            ctx.strokeStyle = 'rgba(47, 129, 247, 0.32)';
            ctx.lineWidth = 1;
            for (let g = 1; g < 4; g++) {
                const gx = (w / 4) * g;
                const gy = (h / 4) * g;
                ctx.beginPath();
                ctx.moveTo(gx, 0);
                ctx.lineTo(gx, h);
                ctx.moveTo(0, gy);
                ctx.lineTo(w, gy);
                ctx.stroke();
            }

            if (sys.isDraggingRect && sys.rectStart && sys.rectCurrent) {
                const x1 = Math.min(sys.rectStart.x, sys.rectCurrent.x) * dpr;
                const y1 = Math.min(sys.rectStart.y, sys.rectCurrent.y) * dpr;
                const rw = Math.abs(sys.rectCurrent.x - sys.rectStart.x) * dpr;
                const rh = Math.abs(sys.rectCurrent.y - sys.rectStart.y) * dpr;

                ctx.fillStyle = sys.rectMode === 'sub' ? 'rgba(248, 81, 73, 0.28)' : 'rgba(255, 158, 0, 0.30)';
                ctx.fillRect(x1, y1, rw, rh);
                ctx.strokeStyle = sys.rectMode === 'sub' ? '#f85149' : '#ffea00';
                ctx.lineWidth = 2 * dpr;
                ctx.strokeRect(x1, y1, rw, rh);
            }
        }

        function emitLaptopTone(freq, durationMs = 200, suppressRx = true) {
            return new Promise((resolve) => {
                if (!sys.audioCtx) {
                    const AudioContextClass = window.AudioContext || window.webkitAudioContext;
                    sys.audioCtx = new AudioContextClass();
                }
                if (sys.audioCtx.state === 'suspended') {
                    sys.audioCtx.resume();
                }

                if (suppressRx) sys.isEmittingTone = true;
                const now = sys.audioCtx.currentTime + 0.01;
                const durSec = durationMs / 1000;

                const osc = sys.audioCtx.createOscillator();
                const gain = sys.audioCtx.createGain();
                osc.type = 'sine';
                osc.frequency.setValueAtTime(freq, now);

                gain.gain.setValueAtTime(0.001, now);
                gain.gain.linearRampToValueAtTime(0.45, now + 0.012);
                gain.gain.setValueAtTime(0.45, now + Math.max(0.018, durSec - 0.012));
                gain.gain.linearRampToValueAtTime(0.001, now + durSec);

                osc.connect(gain);
                gain.connect(sys.audioCtx.destination);
                osc.start(now);
                osc.stop(now + durSec);

                setTimeout(() => {
                    if (suppressRx) sys.isEmittingTone = false;
                    resolve();
                }, durationMs + 25);
            });
        }

            } catch (err) {

        function drawOfflineWaveform(audioBuffer) {
            const canvas = document.getElementById('offline-audio-waveform');
            const ctx = canvas.getContext('2d');
            const width = canvas.width = canvas.clientWidth || 320;
            const height = canvas.height = canvas.clientHeight || 72;
            const samples = audioBuffer.getChannelData(0);
            const samplesPerPixel = Math.max(1, Math.floor(samples.length / width));

            ctx.clearRect(0, 0, width, height);
            ctx.fillStyle = '#02060d';
            ctx.fillRect(0, 0, width, height);
            ctx.strokeStyle = '#a371f7';
            ctx.lineWidth = 1;
            ctx.beginPath();
            for (let x = 0; x < width; x++) {
                const start = x * samplesPerPixel;
                const end = Math.min(samples.length, start + samplesPerPixel);
                let min = 1;
                let max = -1;
                for (let i = start; i < end; i++) {
                    min = Math.min(min, samples[i]);
                    max = Math.max(max, samples[i]);
                }
                ctx.moveTo(x + 0.5, (1 + min) * height / 2);
                ctx.lineTo(x + 0.5, (1 + max) * height / 2);
            }
            ctx.stroke();
        }

        function goertzelPower(frame, frequency, sampleRate) {
            const omega = 2 * Math.PI * frequency / sampleRate;
            const coefficient = 2 * Math.cos(omega);
            let previous = 0;
            let previous2 = 0;
            for (let i = 0; i < frame.length; i++) {
                const current = frame[i] + coefficient * previous - previous2;
                previous2 = previous;
                previous = current;
            }
            const power = previous * previous + previous2 * previous2 -
                coefficient * previous * previous2;
            return Math.max(0, power / (frame.length * frame.length));
        }

        function decodeOfflineSymbols(audioBuffer) {
            const samples = audioBuffer.getChannelData(0);
            const sampleRate = audioBuffer.sampleRate;
            const windowSize = Math.floor(sampleRate * 0.045);
            const hopSize = Math.max(1, Math.floor(sampleRate * 0.025));
            const window = new Float32Array(windowSize);
            for (let i = 0; i < windowSize; i++) {
                window[i] = 0.5 - 0.5 * Math.cos((2 * Math.PI * i) / (windowSize - 1));
            }

            const candidates = [
                { frequency: ACOUSTIC_PROTOCOL.HELLO, type: 'HELLO' },
                { frequency: ACOUSTIC_PROTOCOL.REPEAT, type: 'REPEAT' },
                { frequency: ACOUSTIC_PROTOCOL.EOF, type: 'EOF' },
                ...ACOUSTIC_PROTOCOL.FREQ_TO_CHAR.map(item => ({
                    frequency: item.f,
                    type: 'DATA',
                    ch: item.ch
                }))
            ];
            const detections = [];
            const frame = new Float32Array(windowSize);
            const windowSeconds = windowSize / sampleRate;

            for (let offset = 0; offset + windowSize <= samples.length; offset += hopSize) {
                let mean = 0;
                let squareSum = 0;
                for (let i = 0; i < windowSize; i++) {
                    const value = samples[offset + i];
                    mean += value;
                    squareSum += value * value;
                }
                mean /= windowSize;
                const rms = Math.sqrt(Math.max(0, squareSum / windowSize - mean * mean));
                if (rms < 0.003) continue;

                for (let i = 0; i < windowSize; i++) {
                    frame[i] = (samples[offset + i] - mean) * window[i];
                }

                let bestIndex = -1;
                let bestPower = 0;
                let secondPower = 0;
                for (let i = 0; i < candidates.length; i++) {
                    const power = goertzelPower(frame, candidates[i].frequency, sampleRate);
                    if (power > bestPower) {
                        secondPower = bestPower;
                        bestPower = power;
                        bestIndex = i;
                    } else if (power > secondPower) {
                        secondPower = power;
                    }
                }

                if (bestIndex < 0 || bestPower < 1e-7 ||
                    bestPower < rms * rms * 0.025 ||
                    bestPower < Math.max(1e-9, secondPower) * 1.35) {
                    continue;
                }

                const candidate = candidates[bestIndex];
                detections.push({
                    ...candidate,
                    start: offset / sampleRate,
                    end: offset / sampleRate + windowSeconds
                });
            }

            const events = [];
            let current = null;
            for (const detection of detections) {
                const sameSymbol = current &&
                    current.type === detection.type &&
                    current.ch === detection.ch;
                if (sameSymbol && detection.start - current.end <= 0.085) {
                    current.end = detection.end;
                } else {
                    if (current) events.push(current);
                    current = { ...detection };
                }
            }
            if (current) events.push(current);

            const receptions = [];
            let active = false;
            let payload = '';
            let lastChar = '';
            const savePayload = () => {
                if (payload.length > 0) receptions.push(payload);
            };

            for (const event of events) {
                if (event.type === 'HELLO') {
                    if (active) savePayload();
                    active = true;
                    payload = '';
                    lastChar = '';
                } else if (event.type === 'EOF') {
                    if (active) savePayload();
                    active = false;
                    payload = '';
                    lastChar = '';
                } else if (active && event.type === 'REPEAT' && lastChar) {
                    payload += lastChar;
                } else if (active && event.type === 'DATA') {
                    if (event.ch === 'S') payload = 'S';
                    else payload += event.ch;
                    lastChar = event.ch;
                }
            }
            if (active) savePayload();

            const validReceptions = sys.totalChunks
                ? receptions.filter(value => parseMacroSectorsOrRanges(value, sys.totalChunks).length > 0)
                : receptions;
            return validReceptions.sort((a, b) => b.length - a.length)[0] || '';
        }

        function classifyFrequencySymbol(freqHz) {
            const tol = ACOUSTIC_PROTOCOL.TOLERANCE_HZ;
            if (Math.abs(freqHz - ACOUSTIC_PROTOCOL.HELLO) <= tol) return { type: 'HELLO' };
            if (Math.abs(freqHz - ACOUSTIC_PROTOCOL.REPEAT) <= tol) return { type: 'REPEAT' };
            if (Math.abs(freqHz - ACOUSTIC_PROTOCOL.EOF) <= tol) return { type: 'EOF' };

            for (const item of ACOUSTIC_PROTOCOL.FREQ_TO_CHAR) {
                if (Math.abs(freqHz - item.f) <= tol) {
                    return { type: 'DATA', ch: item.ch };
                }
            }
            return null;
        }

        // Mapea intensidad (0..255) a color térmico para la cascada Waterfall
        function intensityToWaterfallColor(v) {
            if (v < 20) return [3, 7, 16];
            if (v < 70) return [12, 45, 110];
            if (v < 130) return [0, 170, 255];
            if (v < 190) return [0, 255, 95];
            return [255, 238, 0];
        }

        function scanAcousticFFTLoop() {
            if (!sys.audioListening || !sys.analyser) return;

            const canvas = document.getElementById('audio-fft-canvas');
            const ctx = canvas.getContext('2d');
            const w = canvas.width = canvas.clientWidth || 320;
            const h = canvas.height = canvas.clientHeight || 98;
            ctx.clearRect(0, 0, w, h);

            // Si el navegador aún mantiene el AudioContext en estado 'suspended', mostrar aviso interactivo claro
            if (sys.audioCtx && sys.audioCtx.state === 'suspended') {
                ctx.fillStyle = 'rgba(210, 153, 34, 0.16)';
                ctx.fillRect(0, 0, w, h);
                ctx.fillStyle = '#f0ad4e';
                ctx.font = 'bold 11px Consolas, monospace';
                ctx.textAlign = 'center';
                ctx.fillText('⚠️ HAZ CLIC AQUÍ PARA ACTIVAR EL AUDIO DEL NAVEGADOR', w / 2, h / 2 - 4);
                ctx.fillStyle = '#8b949e';
                ctx.font = '10px Consolas, monospace';
                ctx.fillText('(Requerido en páginas HTML locales para habilitar FFT)', w / 2, h / 2 + 12);
                sys.fftLoopId = requestAnimationFrame(scanAcousticFFTLoop);
                return;
            }

            const bufferLen = sys.analyser.frequencyBinCount;
            const data = new Uint8Array(bufferLen);
            sys.analyser.getByteFrequencyData(data);

            const sampleRate = sys.audioCtx.sampleRate || 48000;
            const binHz = sampleRate / sys.analyser.fftSize;

            // Rango visual (200 Hz a 2400 Hz) centrado en la banda donde el micrófono de laptop tiene máxima sensibilidad
            const visMinBin = Math.max(1, Math.floor(200 / binHz));
            const visMaxBin = Math.min(bufferLen - 1, Math.ceil(2400 / binHz));

            // Banda activa del módem FSK en frecuencia vocal humana (630 Hz a 1940 Hz)
            const fskMinBin = Math.floor(630 / binHz);
            const fskMaxBin = Math.min(bufferLen - 1, Math.ceil(1940 / binHz));

            let peakVal = 0;
            let peakBin = fskMinBin;
            let sumVal = 0;
            for (let b = fskMinBin; b <= fskMaxBin; b++) {
                const v = data[b];
                sumVal += v;
                if (v > peakVal) {
                    peakVal = v;
                    peakBin = b;
                }
            }
            const avgNoise = sumVal / Math.max(1, fskMaxBin - fskMinBin + 1);
            const snrDiff = Math.max(0, Math.round(peakVal - avgNoise));

            // Interpolación parabólica sub-bin para máxima precisión en Hz
            let fineBin = peakBin;
            if (peakBin > fskMinBin && peakBin < fskMaxBin) {
                const alpha = data[peakBin - 1];
                const beta = data[peakBin];
                const gamma = data[peakBin + 1];
                const denom = alpha - 2 * beta + gamma;
                if (denom !== 0) {
                    fineBin = peakBin + 0.5 * (alpha - gamma) / denom;
                }
            }
            const dominantFreq = Math.round(fineBin * binHz);

            // Mitad superior (0 .. specH): Analizador de Espectro FFT en vivo
            const specH = Math.floor(h * 0.52);
            const wfY = specH + 2;
            const wfH = h - wfY;

            // Sombrear la zona activa del módem FSK (630 Hz - 1940 Hz)
            const fskStartX = ((fskMinBin - visMinBin) / (visMaxBin - visMinBin)) * w;
            const fskEndX = ((fskMaxBin - visMinBin) / (visMaxBin - visMinBin)) * w;
            ctx.fillStyle = 'rgba(47, 129, 247, 0.12)';
            ctx.fillRect(fskStartX, 0, Math.max(10, fskEndX - fskStartX), specH);

            const numBars = 64;
            const binSpan = Math.max(1, visMaxBin - visMinBin);
            for (let i = 0; i < numBars; i++) {
                const bStart = visMinBin + Math.floor((i / numBars) * binSpan);
                const bEnd = Math.max(bStart + 1, visMinBin + Math.floor(((i + 1) / numBars) * binSpan));
                let maxB = 0;
                for (let b = bStart; b < bEnd && b < bufferLen; b++) {
                    if (data[b] > maxB) maxB = data[b];
                }
                const norm = maxB / 255;
                const isFskZone = bStart >= fskMinBin && bStart <= fskMaxBin;
                if (isFskZone) {
                    ctx.fillStyle = maxB > 50 ? '#00ff41' : '#2f81f7';
                } else {
                    ctx.fillStyle = maxB > 70 ? '#39c5bb' : '#1f4468';
                }
                const barH = Math.max(1, norm * (specH - 4));
                const bx = (i / numBars) * w;
                const bw = Math.max(1.5, (w / numBars) - 1);
                ctx.fillRect(bx, specH - barH, bw, barH);
            }

            // Línea divisoria y etiquetas de banda
            ctx.strokeStyle = '#26354d';
            ctx.beginPath();
            ctx.moveTo(0, specH + 1);
            ctx.lineTo(w, specH + 1);
            ctx.stroke();

            ctx.fillStyle = 'rgba(201, 209, 217, 0.65)';
            ctx.font = '9px Consolas, monospace';
            ctx.textAlign = 'left';
            ctx.fillText('GRAVES (200-630Hz)', 4, 10);
            ctx.textAlign = 'right';
            ctx.fillStyle = '#79c0ff';
            ctx.fillText(`BANDA VOCAL FSK (680-1880Hz) | PICO: ${dominantFreq}Hz`, w - 4, 10);

            // Mitad inferior (wfY .. h): Espectrograma Waterfall 2D en Cascada
            const wfW = waterfallCanvas.width;
            const wfCanvasH = waterfallCanvas.height;
            waterfallCtx.drawImage(waterfallCanvas, 0, 0, wfW, wfCanvasH - 1, 0, 1, wfW, wfCanvasH - 1);

            const rowImg = waterfallCtx.createImageData(wfW, 1);
            const rowPixels = rowImg.data;
            for (let x = 0; x < wfW; x++) {
                const binIdx = Math.min(bufferLen - 1, visMinBin + Math.floor((x / wfW) * binSpan));
                const val = data[binIdx];
                const [r, g, b] = intensityToWaterfallColor(val);
                const p = x * 4;
                rowPixels[p] = r;
                rowPixels[p + 1] = g;
                rowPixels[p + 2] = b;
                rowPixels[p + 3] = 255;
            }
            waterfallCtx.putImageData(rowImg, 0, 0);
            ctx.drawImage(waterfallCanvas, 0, wfY, w, wfH);

            const snrLbl = document.getElementById('rx-snr-lbl');
            if (snrLbl) {
                snrLbl.innerText = `Pico: ${peakVal}/255 | SNR: +${snrDiff} | Gain: ${sys.micGainLevel.toFixed(1)}x`;
            }

            // Demodulación FSK en banda vocal (680-1880 Hz): sensible al celular pero protegida por preámbulo HELLO
            if (!sys.isEmittingTone) {
                const isStrongSignal = (peakVal >= 55 && snrDiff >= 16);
                if (isStrongSignal) {
                    sys.silenceFrames = 0;
                    const sym = classifyFrequencySymbol(dominantFreq);
                    if (sym) {
                        const tag = sym.type === 'DATA' ? `'${sym.ch}'` : sym.type;
                        document.getElementById('fft-freq-lbl').innerText = `${dominantFreq} Hz [${tag}]`;

                        const symKey = sym.type === 'DATA' ? `D_${sym.ch}` : sym.type;
                        if (sys.candidateSymbol === symKey) {
                            sys.candidateHits++;
                        } else {
                            sys.candidateSymbol = symKey;
                            sys.candidateHits = 1;
                        }

                        const requiredHits = sym.type === 'HELLO' ? 3 : 2;
                        if (sys.candidateHits >= requiredHits && sys.lastDecodedSymbol !== symKey) {
                            sys.lastDecodedSymbol = symKey;
                            handleDemodulatedSymbol(sym);
                        }
                    } else {
                        document.getElementById('fft-freq-lbl').innerText = `${dominantFreq} Hz`;
                    }
                } else {
                    sys.silenceFrames++;
                    if (sys.silenceFrames >= 2) {
                        sys.candidateSymbol = null;
                        sys.candidateHits = 0;
                        if (sys.silenceFrames >= 3) {
                            sys.lastDecodedSymbol = null;
                        }
                    }
                    if (sys.silenceFrames > 25 && sys.acousticPhase === 0) {
                        document.getElementById('fft-freq-lbl').innerText = "MIC ACTIVO (ESCUCHANDO)";
                    }
                }
            }

            sys.fftLoopId = requestAnimationFrame(scanAcousticFFTLoop);
        }

        async function handleDemodulatedSymbol(sym) {
            // 1. Tono de inicio HELLO (760 Hz en banda vocal): abre o mantiene abierta la ventana de recepción
            if (sym.type === 'HELLO') {
                sys.rxRleBuffer = "";
                sys.lastSymbolChar = "";
                setSenderAcousticPhaseUI(1);
                clearTimeout(sys.rxSessionTimeout);
                sys.rxSessionTimeout = setTimeout(() => {
                    if (sys.rxRleBuffer.length >= 2) finalizeAcousticRleReception();
                    else setSenderAcousticPhaseUI(0);
                }, 9000);
                return;
            }

            // 2. Solo aceptar datos si estamos formalmente en Fase 1 o 2 (tras haber escuchado HELLO 760Hz)
            if (sym.type === 'DATA' || sym.type === 'REPEAT') {
                if (sys.acousticPhase !== 1 && sys.acousticPhase !== 2) return;

                const ch = (sym.type === 'REPEAT') ? sys.lastSymbolChar : sym.ch;
                if (ch) {
                    if (ch === 'S') {
                        sys.rxRleBuffer = 'S';
                    } else {
                        sys.rxRleBuffer += ch;
                    }
                    sys.lastSymbolChar = ch;
                    setSenderAcousticPhaseUI(2, sys.rxRleBuffer);
                }

                clearTimeout(sys.rxSessionTimeout);
                sys.rxSessionTimeout = setTimeout(() => {
                    if (sys.rxRleBuffer.length >= 2) finalizeAcousticRleReception();
                    else setSenderAcousticPhaseUI(0);
                }, 2200);
                return;
            }

            if (sym.type === 'EOF') {
                if (sys.acousticPhase === 1 || sys.acousticPhase === 2) {
                    clearTimeout(sys.rxSessionTimeout);
                    await finalizeAcousticRleReception();
                }
            }
        }

        async function finalizeAcousticRleReception() {
            const rawStr = sys.rxRleBuffer.replace(/^[, -]+|[, -]+$/g, '');
            sys.rxRleBuffer = "";
            sys.lastSymbolChar = "";

            const liveBuf = document.getElementById('rx-live-buffer');
            if (!rawStr || rawStr.length < 2) {
                setSenderAcousticPhaseUI(0);
                return;
            }

            // Mostrar la trama recibida en el monitor en vivo y en el campo manual sin interrumpir el QR
            document.getElementById('custom-frames').value = rawStr;
            if (liveBuf) liveBuf.innerText = `✅ RECIBIDO: ${rawStr}`;

            if (!sys.totalChunks) {
                setSenderAcousticPhaseUI(3);
                setTimeout(() => {
                    if (sys.acousticPhase === 3) setSenderAcousticPhaseUI(0);
                }, 4000);
                return;
            }

            const decodedMissing = parseMacroSectorsOrRanges(rawStr, sys.totalChunks);
            if (decodedMissing.length > 0) {
                sys.activeIndices = decodedMissing;
                sys.activeSet = new Set(decodedMissing);
                sys.activeIdx = 0;
                setTxMode('priority', true);
                updateTimelineBounds();
                syncCustomFramesInput();
                renderCurrentActiveFrame();

                setSenderAcousticPhaseUI(3);
                document.getElementById('status').innerText = `● BUCLE PRIORITARIO ACTUALIZADO (${decodedMissing.length} FALTANTES DE ${rawStr})`;

                setTimeout(() => {
                    if (sys.acousticPhase === 3) setSenderAcousticPhaseUI(0);
                }, 4000);
            } else {
                setSenderAcousticPhaseUI(0);
            }
        }

        function drawTimelineHeatmap() {
            const canvas = document.getElementById('timeline-heatmap');
            if (!canvas) return;
            const dpr = window.devicePixelRatio || 1;
            const w = (canvas.clientWidth || 400) * dpr;
            const h = (canvas.clientHeight || 22) * dpr;
            if (canvas.width !== w || canvas.height !== h) {
                canvas.width = w;
                canvas.height = h;
            }
            const ctx = canvas.getContext('2d');
            ctx.clearRect(0, 0, w, h);

            if (sys.totalChunks <= 0) return;

            const cols = Math.min(sys.totalChunks, Math.floor(w));
            const sliceW = w / cols;

            for (let c = 0; c < cols; c++) {
                const idxStart = Math.floor((c / cols) * sys.totalChunks);
                const idxEnd = Math.max(idxStart + 1, Math.floor(((c + 1) / cols) * sys.totalChunks));
                let hasActive = false;
                for (let k = idxStart; k < idxEnd; k++) {
                    if (sys.activeSet.has(k)) {
                        hasActive = true;
                        break;
                    }
                }
                if (sys.txMode === 'priority') {
                    ctx.fillStyle = hasActive ? '#ff9e00' : '#0b4f22';
                } else if (sys.txMode === 'fountain') {
                    ctx.fillStyle = '#2f81f7';
                } else {
                    ctx.fillStyle = '#00a82d';
                }
                ctx.fillRect(c * sliceW, 0, Math.ceil(sliceW - (cols < 120 ? 1 : 0)), h);
            }

            if (sys.activeIndices.length > 0 && sys.txMode !== 'fountain') {
                const currentChunk = sys.activeIndices[sys.activeIdx] || 0;
                const cx = (currentChunk / sys.totalChunks) * w;
                ctx.fillStyle = '#ffffff';
                ctx.fillRect(Math.max(0, cx - 2 * dpr), 0, 4 * dpr, h);
            }
        }

        !function initInteractiveHeatmap() {
            const canvas = document.getElementById('timeline-heatmap');
            const tooltip = document.getElementById('heatmap-tooltip');

            function clientXToChunkIndex(clientX) {
                const rect = canvas.getBoundingClientRect();
                const ratio = Math.max(0, Math.min(0.9999, (clientX - rect.left) / rect.width));
                return Math.floor(ratio * sys.totalChunks);
            }

            function applyBrushAt(idx) {
                if (!sys.totalChunks || idx < 0 || idx >= sys.totalChunks) return;
                if (sys.txMode !== 'priority') {
                    setTxMode('priority', true);
                }

                const radius = sys.totalChunks > 500 ? Math.ceil(sys.totalChunks / 200) : 0;
                for (let k = Math.max(0, idx - radius); k <= Math.min(sys.totalChunks - 1, idx + radius); k++) {
                    if (sys.paintAction) {
                        sys.activeSet.add(k);
                    } else if (sys.activeSet.size > 1) {
                        sys.activeSet.delete(k);
                    }
                }

                sys.activeIndices = Array.from(sys.activeSet).sort((a, b) => a - b);
                sys.activeIdx = Math.min(sys.activeIdx, Math.max(0, sys.activeIndices.length - 1));
                updateTimelineBounds();
                syncCustomFramesInput();
                renderCurrentActiveFrame();
            }

            canvas.addEventListener('pointerdown', (e) => {
                if (!sys.totalChunks || sys.activeIndices.length === 0) return;
                const idx = clientXToChunkIndex(e.clientX);

                // Saltar el cursor al bloque activo más cercano sin borrar la cola de transmisión
                let bestPos = 0;
                let bestDist = Infinity;
                for (let i = 0; i < sys.activeIndices.length; i++) {
                    const dist = Math.abs(sys.activeIndices[i] - idx);
                    if (dist < bestDist) {
                        bestDist = dist;
                        bestPos = i;
                    }
                }
                sys.activeIdx = bestPos;
                updateTimelineBounds();
                renderCurrentActiveFrame();
            });

            canvas.addEventListener('pointermove', (e) => {
                if (!sys.totalChunks) return;
                const rect = canvas.getBoundingClientRect();
                const idx = clientXToChunkIndex(e.clientX);
                const isPriority = sys.activeSet.has(idx);

                tooltip.style.display = 'block';
                tooltip.style.left = `${Math.max(45, Math.min(rect.width - 45, e.clientX - rect.left))}px`;
                tooltip.innerText = `Blk #${idx} [${isPriority ? (sys.txMode === 'priority' ? 'En Cola Prioritaria' : 'Activo') : 'Confirmado'}]`;

                if (sys.isPaintingHeatmap) {
                    applyBrushAt(idx);
                }
            });

            window.addEventListener('pointerup', () => {
                sys.isPaintingHeatmap = false;
            });

            canvas.addEventListener('pointerleave', () => {
                tooltip.style.display = 'none';
            });
        }();

        function fillSmallGaps(maxGap = 3) {
            if (!sys.totalChunks || sys.activeIndices.length < 2) return;
            const sorted = [...sys.activeIndices].sort((a, b) => a - b);
            const filled = new Set(sorted);

            for (let i = 0; i < sorted.length - 1; i++) {
                const gap = sorted[i + 1] - sorted[i] - 1;
                if (gap > 0 && gap <= maxGap) {
                    for (let k = sorted[i] + 1; k < sorted[i + 1]; k++) {
                        filled.add(k);
                    }
                }
            }

            sys.activeIndices = Array.from(filled).sort((a, b) => a - b);
            sys.activeIdx = 0;
            setTxMode('priority', true);
            updateTimelineBounds();
            syncCustomFramesInput();
            renderCurrentActiveFrame();
            document.getElementById('status').innerText = `● HUECOS RELLENADOS (${sys.activeIndices.length} BLKS)`;
        }

        function expandActiveIntervals(padding = 2) {
            if (!sys.totalChunks || sys.activeIndices.length === 0) return;
            const expanded = new Set(sys.activeIndices);
            for (const idx of sys.activeIndices) {
                for (let d = -padding; d <= padding; d++) {
                    const target = idx + d;
                    if (target >= 0 && target < sys.totalChunks) {
                        expanded.add(target);
                    }
                }
            }
            sys.activeIndices = Array.from(expanded).sort((a, b) => a - b);
            setTxMode('priority', true);
            updateTimelineBounds();
            syncCustomFramesInput();
            renderCurrentActiveFrame();
            document.getElementById('status').innerText = `● BORDES EXPANDIDOS ±${padding} (${sys.activeIndices.length} BLKS)`;
        }

        function shiftActiveQueue(offset) {
            if (!sys.totalChunks || sys.activeIndices.length === 0) return;

            if (sys.txMode === 'full' || sys.activeIndices.length === sys.totalChunks) {
                sys.activeIdx = (sys.activeIdx + offset + sys.activeIndices.length * 10) % sys.activeIndices.length;
                updateTimelineBounds();
                renderCurrentActiveFrame();
                return;
            }

            const shifted = new Set();
            for (const idx of sys.activeIndices) {
                const nextIdx = (idx + offset + sys.totalChunks * 10) % sys.totalChunks;
                shifted.add(nextIdx);
            }
            sys.activeIndices = Array.from(shifted).sort((a, b) => a - b);
            sys.activeIdx = Math.min(sys.activeIdx, sys.activeIndices.length - 1);
            updateTimelineBounds();
            syncCustomFramesInput();
            renderCurrentActiveFrame();
            document.getElementById('status').innerText = `● COLA DESPLAZADA (${offset > 0 ? '+' : ''}${offset} FRAMES)`;
        }

            } catch (e) { console.error("Error renderFountainParityFrame:", e); }
            triggerPulseLed();
            document.getElementById('frame-badge').innerText = `⚡XOR #${sys.fountainSeed}`;
            document.getElementById('lbl-queue-pos').innerText = `FOUNTAIN (${recipe.length}d)`;
            document.getElementById('lbl-chunk-idx').innerText = `XOR[${recipe.slice(0, 3).join(',')}${recipe.length > 3 ? '..' : ''}]`;
            updateZenHud(`⚡XOR #${sys.fountainSeed}`);
        }

        function renderCurrentActiveFrame() {
            if (!sys.qr || sys.activeIndices.length === 0) return;
            const chunkIdx = sys.activeIndices[sys.activeIdx];
            try {
                sys.qr.makeCode(sys.chunksPayloads[chunkIdx]);
            } catch (e) { console.error("Error renderCurrentActiveFrame:", e); }
            triggerPulseLed();
            document.getElementById('frame-badge').innerText = `BLK #${chunkIdx}`;
            document.getElementById('lbl-queue-pos').innerText = `${sys.activeIdx + 1} / ${sys.activeIndices.length}`;
            document.getElementById('lbl-chunk-idx').innerText = `${chunkIdx} (de ${sys.totalChunks})`;
            document.getElementById('timeline-scrubber').value = sys.activeIdx;
            updateZenHud(`BLK #${chunkIdx}`);
            drawTimelineHeatmap();
            draw2DGridSelector();
        }

        function updateZenHud(frameLabel) {
            const modeLabels = {
                full: 'SISTEMÁTICO + XOR',
                fountain: 'FOUNTAIN PURO (XOR)',
                priority: 'PRIORITY LOOP'
            };
            document.getElementById('zen-hud').innerText =
                `${frameLabel} | COLA ${sys.activeIdx + 1}/${sys.activeIndices.length} | ${sys.fps} FPS | ${modeLabels[sys.txMode] || ''}`;
        }

        function setFPS(fpsVal, fromSlider = false) {
            const fps = Math.max(2, Math.min(30, parseInt(fpsVal, 10) || 12));
            sys.fps = fps;
            const ms = Math.round(1000 / fps);
            document.getElementById('speed-lbl').innerText = `${fps} FPS (${ms} ms)`;
            if (!fromSlider) {
                document.getElementById('fps-range').value = fps;
            }
            document.querySelectorAll('#qr-card .preset-btn').forEach(btn => {
                btn.classList.toggle('active', btn.innerText === `${fps} FPS`);
            });
            updateZenHud(sys.activeIndices.length ? `BLK #${sys.activeIndices[sys.activeIdx] || 0}` : "STANDBY");
            if (sys.running) {
                clearInterval(sys.timer);
                startTimer();
            }
        }

        function adjustFPS(delta) {
            setFPS(sys.fps + delta);
        }

        function startTimer() {
            const intervalMs = Math.round(1000 / sys.fps);
            sys.timer = setInterval(() => {
                if (sys.activeIndices.length === 0) return;
                sys.tickCount++;

                // Cabecera SYN cada 40 ticks en lugar de 30 para reducir parpadeos
                if (sys.tickCount % 40 === 0) {
                    renderHeaderFrame();
                    return;
                }

                // Si quedan muy pocos bloques (<= 3), intercalar paridad XOR cada 2 ticks
                const fewBlocks = sys.activeIndices.length <= 3;
                if (sys.txMode === 'fountain' || (sys.txMode === 'full' && sys.tickCount % 5 === 0) || (fewBlocks && sys.tickCount % 2 === 0)) {
                    renderFountainParityFrame();
                    return;
                }

                renderCurrentActiveFrame();
                sys.activeIdx = (sys.activeIdx + 1) % sys.activeIndices.length;
            }, intervalMs);
        }

        function togglePlay() {
            if (!sys.chunksPayloads.length) return;
            const btn = document.getElementById('run-btn');
            const zBtn = document.getElementById('zen-play');

            if (sys.running) {
                clearInterval(sys.timer);
                sys.running = false;
                btn.innerHTML = "▶ REANUDAR";
                zBtn.innerHTML = "▶";
                document.getElementById('pulse-led').className = 'pulse-led';
                document.getElementById('status').innerText = "● EN PAUSA";
            } else {
                sys.running = true;
                btn.innerHTML = "⏸ PAUSAR";
                zBtn.innerHTML = "⏸";
                document.getElementById('status').innerText = "● TRANSMITIENDO...";
                startTimer();
            }
        }

        function stepFrame(delta) {
            if (!sys.chunksPayloads.length || sys.activeIndices.length === 0) return;
            if (sys.running) togglePlay();
            sys.activeIdx = (sys.activeIdx + delta + sys.activeIndices.length) % sys.activeIndices.length;
            renderCurrentActiveFrame();
        }

        function scrubTimeline(pos) {
            if (!sys.chunksPayloads.length || sys.activeIndices.length === 0) return;
            if (sys.running) togglePlay();
            sys.activeIdx = parseInt(pos, 10);
            renderCurrentActiveFrame();
        }

        function rewind() {
            if (!sys.chunksPayloads.length) return;
            if (sys.running) togglePlay();
            sys.activeIdx = 0;
            sys.tickCount = 0;
            renderHeaderFrame();
        }

        function setTxMode(mode, silent = false) {
            sys.txMode = mode;
            document.getElementById('tx-mode-full').classList.toggle('active', mode === 'full');
            document.getElementById('tx-mode-fountain').classList.toggle('active', mode === 'fountain');
            document.getElementById('tx-mode-priority').classList.toggle('active', mode === 'priority');

            if ((mode === 'full' || mode === 'fountain') && sys.totalChunks > 0 && !silent) {
                sys.activeIndices = Array.from({ length: sys.totalChunks }, (_, i) => i);
                sys.activeIdx = 0;
                updateTimelineBounds();
                syncCustomFramesInput();
                if (mode === 'fountain') renderFountainParityFrame();
                else renderCurrentActiveFrame();
            } else {
                drawTimelineHeatmap();
                draw2DGridSelector();
            }
        }

        // =========================================================================
        // 7. DECODIFICACIÓN DE BITMAP PLANO (664 B) Y BISECCIÓN POR MACRO-SECTORES (S00..S31)
        // =========================================================================
        function triggerFlashAck(missingCount, customTitle = "✅ COLA RECONFIGURADA") {
            // Actualización discreta en barra de estado superior sin mostrar pantallas verdes ni interrumpir el QR
            const statusEl = document.getElementById('status');
            if (statusEl) {
                statusEl.innerText = missingCount > 0
                    ? `● ${customTitle}: ${missingCount} BLOQUES EN COLA`
                    : `● ${customTitle}`;
            }
        }

        function applyMissingBitmapBytes(rawBytes, total) {
            const missingIndices = [];
            for (let i = 0; i < total; i++) {
                const byte = rawBytes[Math.floor(i / 8)] || 0;
                const bit = (byte >> (i % 8)) & 1;
                if (bit === 0) missingIndices.push(i);
            }
            return missingIndices;
        }

        function applyMissingBitmapBase64(b64Bitmap, total) {
            const raw = atob(b64Bitmap);
            const bytes = new Uint8Array(raw.length);
            for (let i = 0; i < raw.length; i++) bytes[i] = raw.charCodeAt(i);
            return applyMissingBitmapBytes(bytes, total);
        }

        // Interpreta identificadores de Macro-Sectores (ej. "S04,09,12" sobre 32 sectores), rangos RLE ("4-12,18") o listas separadas por comas/espacios/saltos de línea
        function parseMacroSectorsOrRanges(str, total) {
            const trimmed = str.trim().toUpperCase();

            // 1. Manejo de Macro-Sectores (S00..S31)
            if (trimmed.startsWith('S')) {
                const numSectors = Math.min(32, Math.max(4, total));
                const sectorSize = Math.ceil(total / numSectors);
                const sectorIds = trimmed.substring(1).split(/[\s,]+/).map(s => parseInt(s.replace(/[^0-9]/g, ''), 10)).filter(n => !isNaN(n));
                const indices = new Set();
                for (const sId of sectorIds) {
                    if (sId >= 0 && sId < numSectors) {
                        const start = sId * sectorSize;
                        const end = Math.min(total - 1, (sId + 1) * sectorSize - 1);
                        for (let i = start; i <= end; i++) indices.add(i);
                    }
                }
                return Array.from(indices).sort((a, b) => a - b);
            }

            // 2. Manejo de Listas separadas por comas, espacios, punto y coma o rangos (0, 1, 4-8, 12...)
            const indices = new Set();
            const tokens = trimmed.split(/[\s,;\n\r]+/);
            for (const token of tokens) {
                if (!token) continue;
                if (token.includes('-')) {
                    const parts = token.split('-');
                    const start = parseInt(parts[0], 10);
                    const end = parseInt(parts[1], 10);
                    if (!isNaN(start) && !isNaN(end)) {
                        for (let i = Math.max(0, start); i <= Math.min(total - 1, end); i++) {
                            indices.add(i);
                        }
                    }
                } else {
                    const num = parseInt(token, 10);
                    if (!isNaN(num) && num >= 0 && num < total) {
                        indices.add(num);
                    }
                }
            }
            return Array.from(indices).sort((a, b) => a - b);
        }

        function processReverseAckPayload(rawText) {
            if (!sys.totalChunks) return false;
            let missing = [];
            const trimmed = rawText.trim();

            try {
                if (trimmed.startsWith('{')) {
                    const obj = JSON.parse(trimmed);
                    if (obj.t === "ACK") {
                        if (typeof obj.z85 === "string") {
                            const bmpBytes = decodeZ85ToBytes(obj.z85, obj.len);
                            missing = applyMissingBitmapBytes(bmpBytes, sys.totalChunks);
                        } else if (typeof obj.b === "string") {
                            missing = applyMissingBitmapBase64(obj.b, sys.totalChunks);
                        } else if (typeof obj.r === "string") {
                            missing = parseMacroSectorsOrRanges(obj.r, sys.totalChunks);
                        }
                    }
                } else if (/^[S0-9,;\-\s]+$/i.test(trimmed)) {
                    missing = parseMacroSectorsOrRanges(trimmed, sys.totalChunks);
                } else {
                    missing = applyMissingBitmapBase64(trimmed, sys.totalChunks);
                }
            } catch (e) {
                console.warn("Formato ACK no reconocido:", e);
                return false;
            }

            if (missing.length === 0) {
                triggerFlashAck(0);
                alert("✅ El receptor reporta 0 bloques faltantes (100% completado).");
                return true;
            }

            sys.activeIndices = missing;
            sys.activeSet = new Set(missing);
            sys.activeIdx = 0;
            setTxMode('priority', true);
            updateTimelineBounds();
            syncCustomFramesInput();
            renderCurrentActiveFrame();
            triggerFlashAck(missing.length);
            return true;
        }

        function applyCustomFrames() {
            if (!sys.totalChunks) return;
            const input = document.getElementById('custom-frames').value.trim();
            if (!input) {
                setTxMode('full');
                return;
            }
            if (processReverseAckPayload(input)) {
                document.getElementById('status').innerText = `● PRIORITY LOOP (${sys.activeIndices.length} FALTANTES)`;
                if (!sys.running) togglePlay();
            } else {
                alert("Formato de macro-sector, rango o bitmap inválido.");
            }
        }

        // Lectura de 1 solo toque desde el portapapeles
        async function pasteFromClipboardAndApply() {
            if (!sys.totalChunks) {
                alert("Primero carga un archivo en el emisor.");
                return;
            }
            try {
                const text = await navigator.clipboard.readText();
                if (!text || !text.trim()) {
                    alert("El portapapeles está vacío.");
                    return;
                }
                document.getElementById('custom-frames').value = text.trim();
                applyCustomFrames();
            } catch (err) {
                const fallback = prompt("Pega aquí la lista de bloques o macro-sectores copiados:");
                if (fallback) {
                    document.getElementById('custom-frames').value = fallback.trim();
                    applyCustomFrames();
                }
            }
        }

        async function openAckScanner() {
            if (!sys.totalChunks) return;
            if (sys.running) togglePlay();

            document.getElementById('ack-modal').classList.add('open');
            document.getElementById('ack-status').innerText = "Iniciando cámara para lectura de Bitmap Plano QR...";

            try {
                if (!sys.ackScanner) {
                    sys.ackScanner = new Html5Qrcode("ack-reader");
                }
                await sys.ackScanner.start(
                    { facingMode: "user" },
                    { fps: 20, qrbox: { width: 250, height: 250 } },
                    (decodedText) => {
                        if (processReverseAckPayload(decodedText)) {
                            document.getElementById('ack-status').innerText = `✅ ¡Bitmap Plano Sincronizado! Retransmitiendo ${sys.activeIndices.length} bloques faltantes.`;
                            closeAckScanner();
                            if (!sys.running) togglePlay();
                        }
                    },
                    () => { }
                );
            } catch (err) {
                document.getElementById('ack-status').innerText = "⚠️ No se pudo acceder a la webcam: " + err;
            }
        }

        async function closeAckScanner() {
            document.getElementById('ack-modal').classList.remove('open');
            if (sys.ackScanner && sys.ackScanner.isScanning) {
                try { await sys.ackScanner.stop(); } catch (e) { }
            }
        }

        function resetZenIdleTimer() {
            if (!document.body.classList.contains('zen-mode')) return;
            document.body.classList.remove('zen-idle');
            clearTimeout(sys.zenIdleTimer);
            sys.zenIdleTimer = setTimeout(() => {
                if (document.body.classList.contains('zen-mode')) {
                    document.body.classList.add('zen-idle');
                }
            }, 3000);
        }

        ['mousemove', 'mousedown', 'touchstart', 'keydown'].forEach(evt => {
            window.addEventListener(evt, resetZenIdleTimer, { passive: true });
        });

        function toggleZen() {
            if (document.body.classList.contains('zen-mode')) exitZen();
            else enterZen();
        }

        function enterZen() {
            if (!sys.totalChunks) return;
            const el = document.documentElement;
            if (el.requestFullscreen) el.requestFullscreen().catch(() => { });
            document.body.classList.add('zen-mode');
            resetZenIdleTimer();
        }

        function exitZen() {
            clearTimeout(sys.zenIdleTimer);
            document.body.classList.remove('zen-idle');
            if (document.exitFullscreen && document.fullscreenElement) {
                document.exitFullscreen().catch(() => { });
            }
            document.body.classList.remove('zen-mode');
        }

        document.addEventListener('fullscreenchange', () => {
            if (!document.fullscreenElement) {
                clearTimeout(sys.zenIdleTimer);
                document.body.classList.remove('zen-idle');
                document.body.classList.remove('zen-mode');
            }
        });

        // =========================================================================
        // 9. WAKE LOCK PERMANENTE EN EMISOR (EVITAR SUSPENSIÓN DE PANTALLA)
        // =========================================================================
        let senderWakeLock = null;
        async function ensureSenderWakeLock() {
            if ('wakeLock' in navigator) {
                try {
                    if (!senderWakeLock || senderWakeLock.released) {
                        senderWakeLock = await navigator.wakeLock.request('screen');
                        senderWakeLock.addEventListener('release', () => {
                            if (document.visibilityState === 'visible') {
                                setTimeout(ensureSenderWakeLock, 300);
                            }
                        });
                    }
                } catch (e) { }
            }
        }
        document.addEventListener('visibilitychange', () => {
            if (document.visibilityState === 'visible') ensureSenderWakeLock();
        });
        ['click', 'keydown', 'pointerdown'].forEach(evt => {
            document.addEventListener(evt, () => {
                if (!senderWakeLock || senderWakeLock.released) ensureSenderWakeLock();
                if (sys.audioCtx && sys.audioCtx.state === 'suspended') {
                    sys.audioCtx.resume().then(() => {
                        const lbl = document.getElementById('fft-freq-lbl');
                        if (lbl && sys.audioListening) lbl.innerText = "MIC ACTIVO";
                    }).catch(() => { });
                }
            }, { passive: true });
        });

            draw2DGridSelector();

            ensureSenderWakeLock();
